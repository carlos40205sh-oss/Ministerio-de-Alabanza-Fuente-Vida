const { onSchedule } = require("firebase-functions/v2/scheduler"); const { onDocumentCreated } = require("firebase-functions/v2/firestore"); const { setGlobalOptions } = require("firebase-functions/v2"); const admin = require("firebase-admin"); admin.initializeApp(); setGlobalOptions({ maxInstances: 2, region: "us-central1" }); const db = admin.firestore(); // Runs every day at 8am Costa Rica time (UTC-6 = 14:00 UTC) exports.notificarEventos = onSchedule("0 14 * * *", async () => { const hoy = new Date(); const manana = new Date(hoy); manana.setDate(manana.getDate() + 1); const fmtDate = d => d.toISOString().slice(0, 10); const hoyStr = fmtDate(hoy); const mananaStr = fmtDate(manana); const eventosSnap = await db.collection("eventos") .where("fecha", "in", [hoyStr, mananaStr]) .get(); if (eventosSnap.empty) return null; const eventos = eventosSnap.docs.map(d => ({ id: d.id, ...d.data() })); const musicosSnap = await db.collection("musicos").get(); const musicos = musicosSnap.docs.map(d => ({ id: d.id, ...d.data() })); for (const ev of eventos) { const esHoy = ev.fecha === hoyStr; const titulo = esHoy ? "🎵 Evento hoy!" : "🎵 Evento mañana!"; const cuerpo = `${ev.titulo}${ev.horaIni ? " a las " + ev.horaIni : ""} · ${ev.tipo}`; const asignadosIds = (ev.asignados || []).map(a => a.musId); const tokens = musicos .filter(m => asignadosIds.includes(m.id) && m.fcmToken) .map(m => m.fcmToken); if (!tokens.length) continue; try { await admin.messaging().sendEachForMulticast({ notification: { title: titulo, body: cuerpo }, webpush: { fcmOptions: { link: "https://carlos40205sh-oss.github.io/Ministerio-de-Alabanza-Fuente-Vida/" } }, data: { eventoId: ev.id }, tokens, }); console.log(`Notificación enviada para: ${ev.titulo} a ${tokens.length} músicos`); } catch (e) { console.error("Error:", e); } } return null; }); // Runs on the last day of each month at 8am Costa Rica time // Runs every Monday at 8am Costa Rica time exports.notificarEventosSemana = onSchedule("0 14 * * 1", async () => { const hoy = new Date(); const finSemana = new Date(hoy); finSemana.setDate(hoy.getDate() + 6); const fmtDate = d => d.toISOString().slice(0, 10); const fmtFecha = f => { const [y,m,d]=f.split('-'); return d+'/'+m; }; const hoyStr = fmtDate(hoy); const finStr = fmtDate(finSemana); // Get events this week const eventosSnap = await db.collection("eventos").get(); const eventos = eventosSnap.docs .map(d => ({ id: d.id, ...d.data() })) .filter(ev => ev.fecha >= hoyStr && ev.fecha <= finStr) .sort((a,b) => a.fecha.localeCompare(b.fecha)); if(!eventos.length) return null; const musicosSnap = await db.collection("musicos").get(); const musicos = musicosSnap.docs.map(d => ({ id: d.id, ...d.data() })); const adminsSnap = await db.collection("admins").get(); const admins = adminsSnap.docs.map(d => ({ id: d.id, ...d.data() })); // Send personalized notification to each musician for (const musico of [...musicos, ...admins]) { if (!musico.fcmToken) continue; const misEventos = eventos.filter(ev => (ev.asignados||[]).some(a => a.musId === musico.id) || (ev.vozAsignados||[]).some(a => a.musId === musico.id) || (ev.sonidoAsignados||[]).some(a => a.musId === musico.id) || (ev.multiAsignados||[]).some(a => a.musId === musico.id) || ev.tipo === 'Ayuno' ); if (!misEventos.length) continue; const resumen = misEventos.map(ev => ev.titulo + ' · ' + fmtFecha(ev.fecha) + (ev.horaIni ? ' ' + ev.horaIni : '') ).join(' | '); try { await admin.messaging().send({ token: musico.fcmToken, notification: { title: '📅 Tus eventos esta semana', body: resumen }, webpush: { fcmOptions: { link: 'https://carlos40205sh-oss.github.io/Ministerio-de-Alabanza-Fuente-Vida/' } } }); console.log('Resumen semanal enviado a:', musico.nombre); } catch(e) { console.error('Error:', e.message); } } return null; }); exports.recordatorioCuotas = onSchedule("0 14 28-31 * *", async () => { const hoy = new Date(); const manana = new Date(hoy); manana.setDate(manana.getDate() + 1); if (manana.getDate() !== 1) return null; const musicosSnap = await db.collection("musicos").get(); const adminsSnap = await db.collection("admins").get(); const tokens = [ ...musicosSnap.docs.map(d => d.data().fcmToken), ...adminsSnap.docs.map(d => d.data().fcmToken) ].filter(Boolean); if (!tokens.length) return null; try { await admin.messaging().sendEachForMulticast({ notification: { title: "💰 Recordatorio de cuota", body: "Hoy es el último día del mes. Si tenés cuotas pendientes, coordiná el pago con la tesorería. ¡Gracias!" }, tokens, }); console.log("Recordatorio de cuotas enviado!"); } catch (e) { console.error("Error:", e); } return null; }); // Triggered when a new anuncio is created in Firestore exports.notificarAnuncio = onDocumentCreated("anuncios/{anuncioId}", async (event) => { const anuncio = event.data.data(); if (!anuncio.notif) return null; // Skip if notifications disabled const musicosSnap = await db.collection("musicos").get(); const adminsSnap = await db.collection("admins").get(); const tokens = [ ...musicosSnap.docs.map(d => d.data().fcmToken), ...adminsSnap.docs.map(d => d.data().fcmToken) ].filter(Boolean); if (!tokens.length) return null; try { await admin.messaging().sendEachForMulticast({ notification: { title: "📢 " + anuncio.titulo, body: anuncio.texto.slice(0, 100) + (anuncio.texto.length > 100 ? "..." : "") }, webpush: { fcmOptions: { link: "https://carlos40205sh-oss.github.io/Ministerio-de-Alabanza-Fuente-Vida/" } }, tokens, }); console.log("Anuncio notificado:", anuncio.titulo); } catch (e) { console.error("Error:", e); } return null; });
